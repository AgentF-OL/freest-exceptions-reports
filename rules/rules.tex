\section{Reduction Rules}

\textbf{Term Reduction} \hfill \fbox{$M \rightarrow_M N$}
\begin{align*}
  & \text{E-TRYINVAL} &~ \tryin{V}{x}{M}{N} &~ \rightarrow_M M\{V/x\}
\end{align*}
\textbf{Configuration Equivalence} \hfill \fbox{$\C~\eq~\D$}

$\C~||~(\D~||~\E)~\eq~(\C~||~\D)~||~\E$
\hfill
$\C~||~\D~\eq~\D~||~\C$
\hfill
$(va)(vb)\C~\eq~(vb)(va)\C$
\begin{center}
  $\C~||~(va)\D~\eq~(va)(\C~||~\D)$,\quad if $a \not\in \fn(\C)$
\end{center}
$\buf{a}{\vv}{b}{\ww}{p}{q}~\eq~\buf{b}{\ww}{a}{\vv}{q}{p}$
\hfill
$()~||~\C~\eq~\C$
\begin{center}
  $(va)(vb)(\buf{a}{\empt}{b}{\empt}{0}{0})~||~\C~\eq~\C$
\end{center}

\textbf{Configuration Reduction} \hfill \fbox{$\C \rightarrow \D$}
\begin{complexalign}
  & \text{E-CHANNEL} \quad\quad\quad\quad \lefteqn{\run{\F}{\channel~@U}                                                    \rightarrow (va)(vb)(\run{\F}{(a,b)})~||~\buf{a}{\empt}{b}{\empt}{0}{0}, \quad \text{where}~a,b~\text{are fresh}} \\                                                   
  & \text{E-FORK}           &~ \run{\F}{\fork~(\lambda ().M)}                                                            &~ \rightarrow \run{\F}{()}~||~M                                                                                     \\
  & \text{E-SEND}           &~ \run{\F}{\send_m~U~a}~||~\buf{a}{\vv}{b}{\ww}{p}{q}                                       &~ \rightarrow \run{\F}{a}~||~\buf{a}{\vv}{b}{\ww \cdot U}{p}{q}                                                     \\
  & \text{E-RECEIVE}        &~ \run{\F}{\receive_m~a}~||~\buf{a}{U \cdot \vv}{b}{\ww}{p}{q}                              &~ \rightarrow \run{\F}{(U,a)}~||~\buf{a}{\vv}{b}{\ww}{p}{q}                                                         \\
  & \text{E-SENDTYPE}       &~ \run{\F}{\sendType~@U~a}~||~\buf{a}{\vv}{b}{\ww}{p}{q}                                    &~ \rightarrow \run{\F}{a}~||~\buf{a}{\vv}{b}{\ww \cdot @U}{p}{q}                                                    \\
  & \text{E-RECEIVETYPE}    &~ \run{\F}{\receiveType~a}~||~\buf{a}{@U \cdot \vv}{b}{\ww}{p}{q}                           &~ \rightarrow \run{\F}{(@U,a)}~||~\buf{a}{\vv}{b}{\ww}{p}{q}                                                        \\
  & \text{E-CLOSE}          &~ \run{\F}{\close~a}~||~\buf{a}{\empt}{b}{\ww}{1}{1}                                        &~ \rightarrow \run{\F}{()}~||~\buf{a}{\empt}{b}{\ww \cdot \eos}{0}{1}                                               \\
  & \text{E-WAIT}           &~ (va)(vb)(\run{\F}{\wait~a}~||~\buf{a}{\eos}{b}{\empt}{1}{0})                              &~ \rightarrow \run{\F}{()}                                                                                          \\
  & \text{E-SELECT}         &~ \run{\F}{\select_m~\lbl~a}~||~\buf{a}{\vv}{b}{\ww}{p}{q}                                  &~ \rightarrow \run{\F}{a}~||~\buf{a}{\vv}{b}{\ww \cdot \lbl}{p}{q}                                                  \\
  & \text{E-CASE} \quad\quad~ \lefteqn{\run{\F}{\caseLabel{i}{n}{m}{\lbl}{a}{M}}~||~\buf{a}{\lbl_j \cdot \vv}{b}{\ww}{p}{q} \rightarrow \run{\F}{M_j}~||~\buf{a_j}{\vv}{b}{\ww}{p}{q}}                                                        \\
  & \text{E-CANCEL}         &~ \run{\F}{\cancel~a}~||~\buf{a}{\vv}{b}{\ww}{p}{q}                                         &~ \rightarrow \run{\F}{()}~||~\zap a~||~\buf{a}{\vv}{b}{\ww}{(p-1)}{q}                                              \\
  & \text{E-ZAP}            &~ \zap a~||~\buf{a}{U \cdot \vv}{b}{\ww}{0}{q}                                              &~ \rightarrow \zap a~||~\zap U~||~\buf{a}{\vv}{b}{\ww}{0}{q}                                                        \\
  & \text{E-RECEIVEZAP}     &~ \run{\F}{\receive_m~a}~||~\zap b~||~\buf{a}{\empt}{b}{\ww}{p}{0}                          &~ \rightarrow \run{\F}{\rraise}~||~\zap b~||~\buf{a}{\empt}{b}{\ww}{(p-1)}{0}                                       \\
  & \text{E-RECEIVETYPEZAP} &~ \run{\F}{\receiveType~a}~||~\zap b~||~\buf{a}{\empt}{b}{\ww}{1}{0}                        &~ \rightarrow \run{\F}{\rraise}~||~\zap a~||~\zap b~||~\buf{a}{\empt}{b}{\ww}{0}{0}                                 \\
  & \text{E-WAITZAP}        &~ \run{\F}{\wait~a}~||~\zap b~||~\buf{a}{\empt}{b}{\ww}{1}{0}                               &~ \rightarrow \run{\F}{\rraise}~||~\zap a~||~\zap b~||~\buf{a}{\empt}{b}{\ww}{0}{0}                                 \\
  & \text{E-CASEZAP} ~ \lefteqn{\run{\F}{\caseLabel{i}{n}{m}{\lbl}{a}{M}}~||~\zap b~||~\buf{a}{\empt}{b}{\ww}{p}{0}         \rightarrow \run{\F}{\rraise}~||~\zap b~||~\buf{a}{\empt}{b}{\ww}{(p-1)}{0}}                                      \\
  & \text{E-TRYINRAISE}     &~ \run{\F}{\tryin{\run{P}{\rraise}}{x}{M}{N}}                                               &~ \rightarrow \run{\F}{N}~||~\zap P                                                                                 \\
  & \text{E-RAISE}          &~ \run{P}{\rraise}                                                                          &~ \rightarrow \zap P                                                                                                \\
  & \text{E-LIFT}           &~ \run{\G}{\C}                                                                              &~ \rightarrow \run{\G}{\D}, \quad \text{if}~\C \rightarrow \D
\end{complexalign}
